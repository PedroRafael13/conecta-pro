### CONSULTAS SQL

```sql
-- Receita real das NFS-e emitidas
SELECT
    to_char(date_trunc('month', data_emissao), 'MM/YYYY') as mes,
    count(*) as nfses,
    round(sum(valor_servicos)::numeric, 2) as receita_bruta,
    round(sum(valor_iss)::numeric, 2) as iss,
    round(sum(valor_servicos - valor_iss)::numeric, 2) as receita_pos_iss
FROM nfses
WHERE status NOT IN ('cancelada', 'rejeitada')
GROUP BY date_trunc('month', data_emissao)
ORDER BY 1 DESC LIMIT 6;

-- Despesas por categoria (DRE gerencial)
SELECT
    category,
    round(sum(ABS(amount))::numeric, 2) as total_ytd,
    round(avg(ABS(amount))::numeric, 2) as media_mensal,
    count(DISTINCT date_trunc('month', transaction_date)) as meses
FROM bank_transactions
WHERE amount < 0
AND transaction_date >= '2026-01-01'
GROUP BY category
ORDER BY total_ytd DESC;
```
